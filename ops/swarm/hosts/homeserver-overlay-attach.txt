# Homeserver: Compose containers on attachable swarm overlays (#3837). Installed to
# /etc/apiary/overlay-attach.txt and read by overlay-attach.py.
# dockerd starts these before its swarm part is up, the start fails and is
# never retried; apiary-overlay-attach.service finishes it. Only containers
# whose dockerd start failed are started, so a deliberately stopped one stays
# down. scripts/tests/test_3837_overlay_attach.py fails if a compose file joins
# a swarm stack overlay that is not listed here.
#
#   start   CONTAINER NETWORK            overlay is in the container's compose file
#   connect CONTAINER NETWORK [ALIAS...] attached after start, never in compose
#   anchor-image IMAGE                   local image for the throwaway attachment
anchor-image busybox:1.38.0

start hp-tftp-relay sensor-dionaea_decoy

# so it stays down unless dockerd itself was starting it
start hp-galah galah-broker_front

# stopped, so it stays down unless dockerd itself was starting it
start ghidra-ollama-1 galah-broker_back
